        // Charts Initialization
        document.addEventListener('DOMContentLoaded', function() {
            const chartTheme = {
                textStyle: { fontFamily: 'Outfit', color: '#bbcabf' },
                grid: { left: '3%', right: '4%', bottom: '3%', containLabel: true }
            };

            const langChart = echarts.init(document.getElementById('languageChart'));
            langChart.setOption({

                yAxis: { type: 'value', splitLine: { lineStyle: { color: 'rgba(255,255,255,0.05)' } } },
                series: [{
                    data: [450, 320, 280, 150, 90, 48],
                    type: 'bar',
                    itemStyle: {
                        color: new echarts.graphic.LinearGradient(0, 0, 0, 1, [
                            { offset: 0, color: '#4edea3' },
                            { offset: 1, color: '#10b981' }
                        ]),
                        borderRadius: [4, 4, 0, 0]
                    }
                }]
            });

            const actChart = echarts.init(document.getElementById('activityChart'));
            actChart.setOption({
                ...chartTheme,
                xAxis: { type: 'category', data: ['1月', '2月', '3月', '4月', '5月', '6月'] },
                yAxis: { type: 'value', splitLine: { lineStyle: { color: 'rgba(255,255,255,0.05)' } } },
                series: [{
                    data: [820, 932, 901, 934, 1290, 1330],
                    type: 'line',
                    smooth: true,
                    lineStyle: { color: '#4cd7f6', width: 3 },
                    areaStyle: {
                        color: new echarts.graphic.LinearGradient(0, 0, 0, 1, [
                            { offset: 0, color: 'rgba(76, 215, 246, 0.3)' },
                            { offset: 1, color: 'transparent' }
                        ])
                    },
                    itemStyle: { color: '#4cd7f6' }
                }]
            });

            window.addEventListener('resize', () => {
                langChart.resize();
                actChart.resize();
            });
        });

        // Interactive states: Selection & Batch Bar
        const checkboxes = document.querySelectorAll('input[type="checkbox"]');
        const batchBar = document.getElementById('batchActionBar');
        const countSpan = document.getElementById('selectedCount');
        const cancelBtn = document.getElementById('cancelSelection');

        const updateBatchBar = () => {
            const checkedCount = Array.from(checkboxes).filter(c => c.checked).length;
            if (checkedCount > 0) {
                countSpan.innerText = checkedCount;
                batchBar.classList.remove('translate-y-32', 'opacity-0');
            } else {
                batchBar.classList.add('translate-y-32', 'opacity-0');
            }
        };

        checkboxes.forEach(cb => {
            cb.addEventListener('change', updateBatchBar);
        });

        cancelBtn.addEventListener('click', () => {
            checkboxes.forEach(cb => cb.checked = false);
            updateBatchBar();
        });

        // Drag & Drop Simulation
        const cards = document.querySelectorAll('.repo-card');
        const sideItems = document.querySelectorAll('nav ul li a');

        cards.forEach(card => {
            card.addEventListener('dragstart', (e) => {
                card.style.opacity = '0.4';
                e.dataTransfer.setData('text/plain', card.querySelector('h3').innerText);
            });
            card.addEventListener('dragend', () => {
                card.style.opacity = '1';
                sideItems.forEach(item => item.classList.remove('drag-over'));
            });
        });

        sideItems.forEach(item => {
            item.addEventListener('dragover', (e) => {
                e.preventDefault();
                item.classList.add('drag-over');
            });
            item.addEventListener('dragleave', () => {
                item.classList.remove('drag-over');
            });
            item.addEventListener('drop', (e) => {
                e.preventDefault();
                const name = e.dataTransfer.getData('text/plain');
                console.log(`Dropped ${name} into ${item.innerText}`);
                item.classList.remove('drag-over');
                // Pulse feedback
                item.classList.add('bg-primary');
                setTimeout(() => item.classList.remove('bg-primary'), 300);
            });
        });

        // Lasso Selection
        let lasso = document.getElementById('lasso');
        let isLassoing = false;
        let startX, startY;

        window.addEventListener('mousedown', (e) => {
            if (e.ctrlKey && e.button === 0) { // Ctrl + Left Click
                isLassoing = true;
                startX = e.clientX;
                startY = e.clientY;
                lasso.style.left = startX + 'px';
                lasso.style.top = startY + 'px';
                lasso.style.width = '0px';
                lasso.style.height = '0px';
                lasso.classList.remove('hidden');
                e.preventDefault();
            }
        });

        window.addEventListener('mousemove', (e) => {
            if (!isLassoing) return;
            let currentX = e.clientX;
            let currentY = e.clientY;

            let width = Math.abs(currentX - startX);
            let height = Math.abs(currentY - startY);
            let left = (currentX < startX) ? currentX : startX;
            let top = (currentY < startY) ? currentY : startY;

            lasso.style.width = width + 'px';
            lasso.style.height = height + 'px';
            lasso.style.left = left + 'px';
            lasso.style.top = top + 'px';

            // Select cards within lasso
            cards.forEach(card => {
                let rect = card.getBoundingClientRect();
                if (rect.left < left + width &&
                    rect.left + rect.width > left &&
                    rect.top < top + height &&
                    rect.top + rect.height > top) {
                    card.querySelector('input').checked = true;
                }
            });
            updateBatchBar();
        });

        window.addEventListener('mouseup', () => {
            isLassoing = false;
            lasso.classList.add('hidden');
        });
    </script>
